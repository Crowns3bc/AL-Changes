async function hunt_rare_variant() { if (character.rip) return; var targets = ["manyeye", "mimic", "paledino"] .map(function (type) { return get_nearest_monster({ type: type }); }) .filter(Boolean) .sort(function (a, b) { return distance(character, a) - distance(character, b); }); var target = targets[0]; if (!target || !can_attack(target)) return; return await attack(target); } hunt_rare_variant().catch(show_json);

function rare_effects() { var target = get_targeted_monster(); var s = (target && target.s) || {}; return { frenzied: !!character.s.frenzied, stoned: !!s.stoned, stonebreak: !!s.stonebreak, cursed: !!s.cursed, exposed: !!s.exposed, sundered: !!s.sundered, }; } show_json(rare_effects());
