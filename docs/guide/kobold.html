async function hunt_kobold() { if (character.rip || character.map !== "ucliffs") return; var target = get_nearest_monster({ type: "kobold" }); if (!target) return; if (!is_in_range(target)) return await move(character.x + (target.x - character.x) / 2, character.y + (target.y - character.y) / 2); if (can_attack(target)) return await attack(target); } hunt_kobold().catch(show_json);
